// Runs the story's plans in node (tsx) and prints them as JSON.
import * as S from "/Users/jmp/gofish-20250316/.claude/worktrees/layout-dsl-proposal/packages/gofish-graphics/stories/_proposals/UserSpaceD3Flow.stories.tsx";

const out: any = {};
for (const a of ["left", "right", "justify", "center"] as const)
  out[a] = Object.fromEntries(S.columnsOf(S.energyNodes, S.energyLinks, a));
const s = S.sankeySizes(S.energyNodes, S.energyLinks, "justify");
out.ky = s.ky;
out.pad = s.pad;
const rel = (p: any) => Object.fromEntries(
  [...S.relaxedTops(s, S.energyLinks, p)].map(([n, y]) => [n, [y, y + s.value(n) * s.ky]])
);
out.relaxed = rel(["right"]);
out.relaxedLR = rel(["left", "right"]);
out.relaxed6 = rel(Array(6).fill(["left", "right"]).flat());
out.insideOut = S.insideOut(S.SERIES);
out.wiggle = S.wiggle(S.SERIES);
out.wiggleInsideOut = S.wiggle(S.insideOut(S.SERIES));
out.force = Object.fromEntries(S.forcePositions());
console.log(JSON.stringify(out));
