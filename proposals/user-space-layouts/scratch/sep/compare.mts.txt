// ── reference: Kennedy's fold (UserSpaceTrees, part 3, byDepth) ──────────
type Tree = { name: string; w?: number; children?: Tree[] };
type Extent = [number, number][];
const R = 7,
  GAP = 10;
const SEP = 2 * R + GAP;
const wOf = (t: Tree) => t.w ?? 2 * R;
const moveExtent = (e: Extent, dx: number): Extent =>
  e.map(([l, r]) => [l + dx, r + dx]);
const mergeExtents = (p: Extent, q: Extent): Extent =>
  Array.from({ length: Math.max(p.length, q.length) }, (_, d) =>
    !p[d] ? q[d] : !q[d] ? p[d] : [p[d][0], q[d][1]]
  );
const flip = (e: Extent): Extent => e.map(([l, r]) => [-r, -l]);
const fitEdges = (p: Extent, q: Extent): number =>
  Math.max(0, ...p.slice(0, q.length).map(([, r], d) => r - q[d][0] + GAP));
const packLeft = (es: Extent[]): number[] => {
  let acc: Extent = [];
  return es.map((e) => {
    const x = fitEdges(acc, e);
    acc = mergeExtents(acc, moveExtent(e, x));
    return x;
  });
};
const packRight = (es: Extent[]) =>
  packLeft(es.map(flip).reverse())
    .map((x) => -x)
    .reverse();
const centered = (xs: number[]) =>
  xs.map((x) => x - (xs[0] + xs[xs.length - 1]) / 2);
const leftRule = (es: Extent[]) => centered(packLeft(es));
const meanRule = (es: Extent[]) => {
  const [l, r] = [packLeft(es), packRight(es)];
  return l.map((x, i) => (x + r[i]) / 2);
};
type Placed = { offsets: number[]; extent: Extent; kids: Placed[] };
const designBy =
  (rule: (es: Extent[]) => number[]) =>
  (t: Tree): Placed => {
    const kids = (t.children ?? []).map(designBy(rule));
    const offsets = rule(kids.map((k) => k.extent));
    const extent: Extent = [
      [-wOf(t) / 2, wOf(t) / 2],
      ...kids.reduce<Extent>(
        (acc, k, i) => mergeExtents(acc, moveExtent(k.extent, offsets[i])),
        []
      ),
    ];
    return { offsets, extent, kids };
  };
// TreeContours' `design` (centers, SEP): part 1 of UserSpaceTrees.
const fitCenters = (p: Extent, q: Extent): number =>
  Math.max(0, ...p.slice(0, q.length).map(([, r], d) => r - q[d][0] + SEP));
const design = (t: Tree): Placed => {
  const kids = (t.children ?? []).map(design);
  const pl = (es: Extent[]) => {
    let acc: Extent = [];
    return es.map((e) => {
      const x = fitCenters(acc, e);
      acc = mergeExtents(acc, moveExtent(e, x));
      return x;
    });
  };
  const l = pl(kids.map((k) => k.extent));
  const r = pl(kids.map((k) => flip(k.extent)).reverse())
    .map((x) => -x)
    .reverse();
  const offsets = l.map((x, i) => (x + r[i]) / 2);
  const extent = kids.reduce<Extent>(
    (acc, k, i) => mergeExtents(acc, moveExtent(k.extent, offsets[i])),
    []
  );
  return { offsets, extent: [[0, 0], ...extent], kids };
};

/** Absolute x of every node, root at 0. */
const absX = (d: { offsets: number[]; kids: any[] }, x = 0, out: number[] = []) => {
  out.push(x);
  d.kids.forEach((k, i) => absX(k, x + d.offsets[i], out));
  return out;
};
const maxDiff = (a: number[], b: number[]) => {
  if (a.length !== b.length) throw new Error("length mismatch");
  return Math.max(0, ...a.map((v, i) => Math.abs(v - b[i])));
};

type Link = { source: string; target: string };
const layers = (nodes: string[], links: Link[]): Map<string, number> => {
  const rank = new Map<string, number>();
  const visit = (id: string): number => {
    if (!rank.has(id)) {
      const preds = links.filter((l) => l.target === id);
      rank.set(id, Math.max(-1, ...preds.map((l) => visit(l.source))) + 1);
    }
    return rank.get(id)!;
  };
  nodes.forEach(visit);
  return rank;
};
const rankDiff = (a: Map<string, number>, b: Map<string, number>, nodes: string[]) =>
  nodes.filter((id) => a.get(id) !== b.get(id)).length;

const isDummy = (id: string) => id.includes(">");
const heightOf = (id: string) => (isDummy(id) ? 8 : 22);

const splitLongEdges = (nodes: string[], links: Link[]) => {
  const rank = layers(nodes, links);
  const segments: Link[] = [];
  for (const { source, target } of links) {
    let prev = source;
    for (let k = rank.get(source)! + 1; k < rank.get(target)!; k++) {
      const dummy = `${source}>${target}@${k}`;
      rank.set(dummy, k);
      segments.push({ source: prev, target: dummy });
      prev = dummy;
    }
    segments.push({ source: prev, target });
  }
  const rows: string[][] = [];
  for (const id of new Set([...nodes, ...rank.keys()])) (rows[rank.get(id)!] ??= []).push(id);
  return { rows, links: segments };
};
const crossings = (rows: string[][], links: Link[]): number => {
  const at = new Map(rows.flatMap((row) => row.map((id, i) => [id, i])));
  const layer = new Map(rows.flatMap((row, k) => row.map((id) => [id, k])));
  let n = 0;
  links.forEach((a, i) =>
    links.slice(i + 1).forEach((b) => {
      const sameGap = layer.get(a.source) === layer.get(b.source);
      const ds = at.get(a.source)! - at.get(b.source)!;
      const dt = at.get(a.target)! - at.get(b.target)!;
      if (sameGap && ds * dt < 0) n++;
    })
  );
  return n;
};
const order = (rows: string[][], links: Link[], sweeps = 4): string[][] => {
  let best = rows;
  for (let s = 0; s < sweeps; s++) {
    const down = s % 2 === 0;
    const ks = rows.map((_, k) => k);
    for (const k of down ? ks.slice(1) : ks.reverse().slice(1)) {
      const at = new Map(rows.flatMap((row) => row.map((id, i) => [id, i])));
      const bary = (id: string) => {
        const ns = links
          .filter((l) => (down ? l.target : l.source) === id)
          .map((l) => at.get(down ? l.source : l.target)!);
        return ns.length ? ns.reduce((a, b) => a + b) / ns.length : at.get(id)!;
      };
      rows = rows.map((row, i) => (i === k ? [...row].sort((a, b) => bary(a) - bary(b)) : row));
    }
    if (crossings(rows, links) < crossings(best, links)) best = rows;
  }
  return best;
};
const straighten = (rows: string[][], links: Link[]): Map<string, number> => {
  const y = new Map<string, number>();
  const mean = (xs: number[]) => xs.reduce((a, b) => a + b) / xs.length;
  rows.forEach((row, k) => {
    const sep = row.map((id, i) => (i ? (heightOf(row[i - 1]) + heightOf(id)) / 2 + GAP_Y : 0));
    const want = row.map((id, i) =>
      k === 0
        ? sep.slice(0, i + 1).reduce((a, b) => a + b)
        : mean(links.filter((l) => l.target === id).map((l) => y.get(l.source)!))
    );
    const down = [...want];
    for (let i = 1; i < row.length; i++) down[i] = Math.max(want[i], down[i - 1] + sep[i]);
    const up = [...want];
    for (let i = row.length - 2; i >= 0; i--) up[i] = Math.min(want[i], up[i + 1] - sep[i + 1]);
    row.forEach((id, i) => y.set(id, (down[i] + up[i]) / 2));
  });
  return y;
};
