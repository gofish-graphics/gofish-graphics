+/**
+ * A node whose arrangement is a user function of its children, called in
+ * its layout step: `body(children)` returns an arrangement built with the
+ * ordinary operators, which is resolved, made a child of this node, and laid
+ * out in the room this node is given, in that call. A test of the engine, not
+ * the API a user would write (`layout(children, body)`).
+ */
+function userLayout(
+  children: GoFishNode[],
+  body: (children: GoFishNode[]) => MarkChild
+): GoFishNode {
+  const calls: string[] = [];
+  const node: GoFishNode = new GoFishNode(
+    {
+      type: "user-layout",
+      // The arrangement does not exist before layout, so this node reports
+      // the type its children have side by side: none on either axis
+      // (pixel sizes only, which is what this test lays out).
+      resolveUnderlyingSpace: (spaces) => spaces[0],
+      layout: (size, scales, _children, self) => {
+        calls.push("layout");
+        // 1. Build: the public operators over existing nodes, resolved now.
+        const arranged = build(body(children));
+        // 2. What a render root does for a fresh tree, for this subtree:
+        //    the session, the axis-scope hooks (spread installs its
+        //    constraints here), refs, and the type walk.
+        arranged.setRenderSession(self.getRenderSession());
+        arranged.resolveAliases();
+        resolveReferences(arranged);
+        arranged.resolveUnderlyingSpace();
+        arranged.resolveEmbedding();
+        self.children = [arranged];
+        arranged.parent = self;
+        // 3. Lay it out, here, in this node's room.
+        const placed = arranged.layout(size, scales);
+        placeUnplacedChild(placed);
+        return {
+          intrinsicDims: placed.dims.map(({ min, size }) => ({ min, size })),
+          transform: { translate: [undefined, undefined] },
+        } as any;
+      },
+    },
+    children as GoFishAST[]
+  );
+  (node as any).__calls = calls;
+  return node;
+}

+/**
+ * A value, or a promise of it. Building a node is synchronous; it becomes a
+ * promise only where something under it is truly asynchronous: a Python
+ * callback over the bridge, an image whose size is not known yet, a `derive`
+ * that awaits. The helpers below compose such values without adding a step
+ * where there is none, so a build over plain values and nodes runs to the
+ * end in the call that started it.
+ */
+export type MaybePromise<T> = T | PromiseLike<T>;
+
+/** `f` of the value of `x`: now when `x` is a value, after it when it is a
+ *  promise. */
+export function andThen<T, U>(
+  x: MaybePromise<T>,
+  f: (value: T) => MaybePromise<U>
+): MaybePromise<U> {
+  return isThenable(x) ? Promise.resolve(x).then(f) : f(x);
+}
+
+/** The values of `xs`, as `Promise.all` gives them: now when none is a
+ *  promise. */
+export function allOf<T>(xs: readonly MaybePromise<T>[]): MaybePromise<T[]> {
+  return xs.some(isThenable) ? Promise.all(xs) : (xs as T[]);
+}
+
+/** `f` over `items` one at a time, each after the one before it has its
+ *  value (an effect of one, such as a name it registers, is there for the
+ *  next): now when every call returns a value. */
+export function mapInOrder<T, U>(
+  items: readonly T[],
+  f: (item: T, i: number) => MaybePromise<U>
+): MaybePromise<U[]> {
+  const out: U[] = [];
+  const from = (start: number): MaybePromise<U[]> => {
+    for (let i = start; i < items.length; i++) {
+      const r = f(items[i], i);
+      if (isThenable(r))
+        return Promise.resolve(r).then((v) => {
+          out.push(v);
+          return from(i + 1);
+        });
+      out.push(r);
+    }
+    return out;
+  };
+  return from(0);
+}
+
+/**
+ * The value of a step the synchronous core runs (resolution, chrome, layout,
+ * placement). The core never waits: whatever is asynchronous is resolved
+ * before it starts (a chart's data, its `derive` calls and Python callbacks,
+ * fonts, images). A step that still reaches something asynchronous is an
+ * error, named by `where`.
+ */
+export function now<T>(x: MaybePromise<T>, where: string): T {
+  if (!isThenable(x)) return x;
+  // Nobody waits for it: keep its own failure from surfacing as unhandled.
+  x.then(undefined, () => {});
+  throw new Error(
+    `${where} reached an asynchronous step (a Python callback, an image ` +
+      `load, an async function) inside the synchronous layout core. ` +
+      `Resolve it before rendering: build the node first (await it), then ` +
+      `use the node.`
+  );
+}
