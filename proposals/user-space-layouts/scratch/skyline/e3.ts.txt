// Built-in replica: nearest free spot (base, or any interval end) on the allowed side.
function nearestFree(side: "start" | "middle") {
  const ys: number[] = [];
  xs.forEach((x, i) => {
    const iv: [number, number][] = [];
    for (let j = 0; j < i; j++) {
      const dx = x - xs[j], dr = 2 * R + PAD;
      if (Math.abs(dx) >= dr) continue;
      const dy = Math.sqrt(dr * dr - dx * dx);
      iv.push([ys[j] - dy, ys[j] + dy]);
    }
    const base = side === "start" ? R : 0;
    const cands = [base, ...iv.flat()].filter((c) => (side === "start" ? c >= base - EPS : true));
    const free = (c: number) => iv.every(([lo, hi]) => !(c > lo + EPS && c < hi - EPS));
    let best = NaN, bd = Infinity;
    for (const c of cands) if (free(c) && Math.abs(c - base) < bd - 1e-12) (best = c), (bd = Math.abs(c - base));
    ys.push(best);
  });
  return ys;
}

// One-sided skyline: lane = x, depth = y, floor = r (the dot sits on the line).
const sky1 = skyline(xs.map((x) => ({ x, r: R })), cartesian, { lane: (d) => d.x, floor: R, pad: PAD }).map((p) => p.y);
const nf1 = nearestFree("start");

// Two-sided: each dot slides toward the line from above and from below; it
// takes the side with the smaller |y| (up on a tie).
function twoSided() {
  const ys: number[] = [];
  xs.forEach((x, i) => {
    let up = 0, down = 0;
    for (let j = 0; j < i; j++) {
      const dx = x - xs[j], dr = 2 * R + PAD;
      if (Math.abs(dx) >= dr) continue;
      const dy = Math.sqrt(dr * dr - dx * dx);
      up = Math.max(up, ys[j] + dy);
      down = Math.min(down, ys[j] - dy);
    }
    ys.push(up <= -down ? up : down);
  });
  return ys;
}
